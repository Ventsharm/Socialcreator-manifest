match /users/{uid}/budgetPushTokens/{tokenId} {
  allow read, delete: if request.auth != null && request.auth.uid == uid;
  allow create, update: if request.auth != null && request.auth.uid == uid
    && tokenId.matches('^[a-f0-9]{64}$')
    && request.resource.data.keys().hasOnly(['token', 'updatedAt'])
    && request.resource.data.token is string
    && request.resource.data.token.size() > 20
    && request.resource.data.token.size() <= 4096
    && request.resource.data.updatedAt is number;
}

// Server-only deduplication records; the sender uses the Admin SDK.
match /budgetReminderDeliveries/{deliveryId} {
  allow read, write: if false;
}
